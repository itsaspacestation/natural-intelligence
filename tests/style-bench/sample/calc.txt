procedure sum(items):
    total = 0
    for each item in items:
        total = total + item
    return total

procedure average(items):
    if count(items) == 0:
        return 0
    return sum(items) / count(items)

procedure max_value(items):
    best = items[0]
    for each item in items:
        if item > best:
            best = item
    return best
